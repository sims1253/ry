to_seconds <- function(seconds) {
  if (is.numeric(seconds) || all(is.na(seconds))) {

}

to_seconds <- function(seconds) {
  if (is.numeric(seconds) || (length(seconds) > 0 && all(is.na(seconds)))) {
    sqrt(seconds)
  }
}
